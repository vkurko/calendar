## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, ResourceTimeline, Interaction} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const resources = [/* your resources */];
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [ResourceTimeline, Interaction], {
    view: 'resourceTimelineWeek',
    editable: true,
    slotMinTime: '08:00',
    slotMaxTime: '20:00',
    slotDuration: '01:00',
    slotWidth: 72,
    resources,
    events,
    eventDrop: ({event, newResource}) => {
        console.log('eventDrop', event.title, newResource ? '→ ' + newResource.title : '');
    }
});
```
