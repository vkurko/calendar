<TabItem label="Vanilla JS">

```js
import {createCalendar, TimeGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const today = new Date();
const addDays = (date, days) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days);
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [TimeGrid], {
    view: 'timeGridWeek',
    scrollTime: '08:00',
    headerToolbar: {start: 'title', center: '', end: 'addEvent nextMonth today prev,next'},
    customButtons: {
        addEvent: {
            text: '+ Event',
            click: () => ec.addEvent({start: today, end: today, allDay: true, title: 'Added by the button'})
        },
        nextMonth: {
            text: 'In a month',
            click: () => ec.gotoDate(addDays(ec.getOption('date'), 30))
        }
    },
    events
});
```

<TabItem label="React">

```jsx
import {useRef} from 'react';
import {EventCalendar, TimeGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [TimeGrid];
const today = new Date();
const addDays = (date, days) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days);
const events = [/* your events */];

export default function App() {
    const calendar = useRef(null);
    const options = {
        view: 'timeGridWeek',
        scrollTime: '08:00',
        headerToolbar: {start: 'title', center: '', end: 'addEvent nextMonth today prev,next'},
        customButtons: {
            addEvent: {
                text: '+ Event',
                click: () => calendar.current.addEvent({start: today, end: today, allDay: true, title: 'Added by the button'})
            },
            nextMonth: {
                text: 'In a month',
                click: () => calendar.current.gotoDate(addDays(calendar.current.getOption('date'), 30))
            }
        },
        events
    };

    return (
        <EventCalendar
            ref={calendar}
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component, viewChild} from '@angular/core';
import {EventCalendar, EcContent, TimeGrid, type Calendar} from '@event-calendar/angular';

const today = new Date();
const addDays = (date, days) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days);
const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    calendar = viewChild.required(EventCalendar);
    plugins = [TimeGrid];
    options: Calendar.Options = {
        view: 'timeGridWeek',
        scrollTime: '08:00',
        headerToolbar: {start: 'title', center: '', end: 'addEvent nextMonth today prev,next'},
        customButtons: {
            addEvent: {
                text: '+ Event',
                click: () => this.calendar().addEvent({start: today, end: today, allDay: true, title: 'Added by the button'})
            },
            nextMonth: {
                text: 'In a month',
                click: () => this.calendar().gotoDate(addDays(this.calendar().getOption('date'), 30))
            }
        },
        events
    };
}
```
