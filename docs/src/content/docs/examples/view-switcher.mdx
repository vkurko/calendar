## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, DayGrid, TimeGrid, List} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const today = new Date();
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [DayGrid, TimeGrid, List], {
    view: 'timeGridWeek',
    headerToolbar: {
        start: 'title',
        center: '',
        end: 'dayGridMonth,timeGridWeek,timeGridDay,listWeek today prev,next'
    },
    // The texts of the buttons
    buttonText: text => ({
        ...text, today: 'Today', dayGridMonth: 'Month', timeGridWeek: 'Week', timeGridDay: 'Day', listWeek: 'List'
    }),
    scrollTime: '08:00',
    events
});
```

import '@event-calendar/react/index.css';

const plugins = [DayGrid, TimeGrid, List];

    const options = {

            end: 'dayGridMonth,timeGridWeek,timeGridDay,listWeek today prev,next'
        },
        // The texts of the buttons

            ...text, today: 'Today', dayGridMonth: 'Month', timeGridWeek: 'Week', timeGridDay: 'Day', listWeek: 'List'

        events
    };

    return (
        <EventCalendar

const today = new Date();

    };
}
