A classic month grid. With `dayMaxEvents` the rows keep the same height and the events that do not fit are collected under the “+more” link.

<Example client:only="svelte" slug="month-view" />

## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, DayGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [DayGrid], {
    view: 'dayGridMonth',
    // The limit is the height of the day cells, so the calendar needs a height
    height: '720px',
    dayMaxEvents: true,
    events
});
```

</TabItem>
<TabItem label="Svelte">


</TabItem>
<TabItem label="React">

```jsx
import {EventCalendar, DayGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [DayGrid];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'dayGridMonth',
        // The limit is the height of the day cells, so the calendar needs a height
        height: '720px',
        dayMaxEvents: true,
        events
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts

import {EventCalendar, EcContent, DayGrid, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [DayGrid];
    options: Calendar.Options = {
        view: 'dayGridMonth',
        // The limit is the height of the day cells, so the calendar needs a height

        dayMaxEvents: true,
        events
    };
}
```
