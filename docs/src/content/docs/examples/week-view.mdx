## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, TimeGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [TimeGrid], {
    view: 'timeGridWeek',
    nowIndicator: true,
    scrollTime: '08:00',
    events
});
```

</TabItem>
<TabItem label="Svelte">

```svelte
<script>
    import {Calendar, TimeGrid} from '@event-calendar/core';

    let events = [/* your events */];
    let options = $state({
        view: 'timeGridWeek',
        nowIndicator: true,
        scrollTime: '08:00',
        events
    });
</script>

<Calendar plugins={[TimeGrid]} {options} />
```

</TabItem>
<TabItem label="React">

```jsx
import {EventCalendar, TimeGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [TimeGrid];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'timeGridWeek',
        nowIndicator: true,
        scrollTime: '08:00',
        events
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, TimeGrid, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [TimeGrid];
    options: Calendar.Options = {
        view: 'timeGridWeek',
        nowIndicator: true,
        scrollTime: '08:00',
        events
    };
}
```
