## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, DayGrid, Interaction} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [DayGrid, Interaction], {
    view: 'dayGridMonth',
    events,
    dateClick: ({date}) => {
        console.log('dateClick', date);
        ec.addEvent({start: date, end: date, allDay: true, title: 'Added by click', color: '#14b8a6'});
    },
    eventClick: ({event}) => {
        console.log('eventClick', event.title);
        ec.removeEventById(event.id);
    }
});
```

<TabItem label="React">

```jsx
import {useRef} from 'react';
import {EventCalendar, DayGrid, Interaction} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [DayGrid, Interaction];
const events = [/* your events */];

export default function App() {
    const calendar = useRef(null);
    const options = {
        view: 'dayGridMonth',
        events,
        dateClick: ({date}) => {
            console.log('dateClick', date);
            calendar.current.addEvent({start: date, end: date, allDay: true, title: 'Added by click', color: '#14b8a6'});
        },
        eventClick: ({event}) => {
            console.log('eventClick', event.title);
            calendar.current.removeEventById(event.id);
        }
    };

    return (
        <EventCalendar
            ref={calendar}
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component, viewChild} from '@angular/core';
import {EventCalendar, EcContent, DayGrid, Interaction, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    calendar = viewChild.required(EventCalendar);
    plugins = [DayGrid, Interaction];
    options: Calendar.Options = {
        view: 'dayGridMonth',
        events,
        dateClick: ({date}) => {
            console.log('dateClick', date);
            this.calendar().addEvent({start: date, end: date, allDay: true, title: 'Added by click', color: '#14b8a6'});
        },
        eventClick: ({event}) => {
            console.log('eventClick', event.title);
            this.calendar().removeEventById(event.id);
        }
    };
}
```
