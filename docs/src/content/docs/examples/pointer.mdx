With `pointer` the slot under the cursor is highlighted, which helps to aim before a click.

<Example client:only="svelte" slug="pointer" />

## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, TimeGrid, Interaction} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [TimeGrid, Interaction], {
    view: 'timeGridWeek',
    pointer: true,
    scrollTime: '08:00',
    events,
    dateClick: ({date}) => console.log('dateClick', date)
});
```

</TabItem>
<TabItem label="Svelte">

```svelte
<script>
    import {Calendar, TimeGrid, Interaction} from '@event-calendar/core';
```

</TabItem>
<TabItem label="React">

```jsx
import {EventCalendar, TimeGrid, Interaction} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [TimeGrid, Interaction];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'timeGridWeek',
        pointer: true,
        scrollTime: '08:00',
        events,
        dateClick: ({date}) => console.log('dateClick', date)
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, TimeGrid, Interaction, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [TimeGrid, Interaction];
    options: Calendar.Options = {
        view: 'timeGridWeek',
        pointer: true,
        scrollTime: '08:00',
        events,
        dateClick: ({date}) => console.log('dateClick', date)
    };
}
```
