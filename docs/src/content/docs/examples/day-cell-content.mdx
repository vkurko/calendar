## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, DayGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [DayGrid], {
    view: 'dayGridMonth',
    events,
    dayCellContent: ({date, isToday}) => ({
        html: `
            <span class="weekday">${date.toLocaleDateString('en', {weekday: 'short'})}</span>
            <span class="${isToday ? 'number today' : 'number'}">${date.getDate()}</span>
        `
    })
});
```

</TabItem>
<TabItem label="Svelte">

<script>
    import {Calendar, DayGrid} from '@event-calendar/core';

    let events = [/* your events */];
    let options = $state({
        view: 'dayGridMonth',
        events
    });

<Calendar plugins={[DayGrid]} {options}>
    {#snippet dayCellContent({date, isToday})}
        <span class="weekday">{date.toLocaleDateString('en', {weekday: 'short'})}</span>

    {/snippet}
</Calendar>

<TabItem label="React">

```jsx
import {EventCalendar, DayGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [DayGrid];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'dayGridMonth',
        events
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
            dayCellContent={({date, isToday}) => (
                <>
                    <span className="weekday">{date.toLocaleDateString('en', {weekday: 'short'})}</span>
                    <span className={isToday ? 'number today' : 'number'}>{date.getDate()}</span>
                </>
            )}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, DayGrid, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `
        <event-calendar [plugins]="plugins" [options]="options">
            <ng-template ecContent="dayCellContent" let-date="date" let-isToday="isToday">
                <span class="weekday">{{ date.toLocaleDateString('en', {weekday: 'short'}) }}</span>
                <span class="number" [class.today]="isToday">{{ date.getDate() }}</span>
            </ng-template>
        </event-calendar>`
})
export class App {
    plugins = [DayGrid];
    options: Calendar.Options = {
        view: 'dayGridMonth',
        events
    };
}
```
