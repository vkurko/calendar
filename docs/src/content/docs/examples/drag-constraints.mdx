The events cannot be dropped or resized into the weekend, and a selection cannot start there.

<Example client:only="svelte" slug="drag-constraints" />

## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, TimeGrid, Interaction} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [TimeGrid, Interaction], {
    view: 'timeGridWeek',
    editable: true,
    selectable: true,
    scrollTime: '08:00',
    events,
    dragConstraint: ({event}) => ![0, 6].includes(event.start.getDay()),
    resizeConstraint: ({event}) => ![0, 6].includes(event.end.getDay()),
    selectConstraint: ({start}) => ![0, 6].includes(start.getDay())
});
```
