## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, List} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [List], {
    view: 'listWeek',
    events
});
```

</TabItem>
<TabItem label="Svelte">

```svelte
<script>
    import {Calendar, List} from '@event-calendar/core';

    let events = [/* your events */];
    let options = $state({
        view: 'listWeek',
        events
    });
</script>

<Calendar plugins={[List]} {options} />
```

</TabItem>
<TabItem label="React">

```jsx
import {EventCalendar, List} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [List];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'listWeek',
        events
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
        />
    );
}
```

</TabItem>

</TabItem>
<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, List, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [List];
    options: Calendar.Options = {
        view: 'listWeek',
        events
    };
}
```

</TabItem>
