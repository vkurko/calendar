<TabItem label="Vanilla JS">

```js
import {createCalendar, ResourceTimeGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const today = new Date();
const people = [/* your resources */];
const meetings = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [ResourceTimeGrid], {
    view: 'resourceTimeGridDay',
    height: 'auto',
    slotMinTime: '08:00',
    slotMaxTime: '16:00',
    slotHeight: 40,
    allDaySlot: false,
    nowIndicator: true,
    buttonText: text => ({...text, today: 'Today'}),
    // The cards have their own colors
    eventBackgroundColor: 'transparent',
    // Each resource has a role and a photo in its extendedProps
    resources: people,
    // Each event has a type, a room and the attendees in its extendedProps
    events: meetings,
    eventContent: ({event, timeText}) => ({
        html: `
            <div class="card ${event.extendedProps.type}">
                <div class="card-title">${event.title}</div>
                <div class="card-meta">${timeText} · ${event.extendedProps.room}</div>
                <div class="card-footer">
                    ${event.extendedProps.attendees.map(person => `<img src="${person.photo}" alt="${person.name}" title="${person.name}">`).join('')}
                    <span class="card-tag">${event.extendedProps.type}</span>
                </div>
            </div>
        `
    }),
    resourceLabelContent: ({resource}) => ({
        html: `
            <div class="member">
                <img src="${resource.extendedProps.photo}" alt="">
                <b>${resource.title}</b>
                <small>${resource.extendedProps.role}</small>
            </div>
        `
    })
});
```

import {EventCalendar, ResourceTimeGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [ResourceTimeGrid];
const today = new Date();
const people = [/* your resources */];
const meetings = [/* your events */];

export default function App() {
    const options = {
        view: 'resourceTimeGridDay',
        height: 'auto',
        slotMinTime: '08:00',
        slotMaxTime: '16:00',
        slotHeight: 40,
        allDaySlot: false,
        nowIndicator: true,
        buttonText: text => ({...text, today: 'Today'}),
        // The cards have their own colors
        eventBackgroundColor: 'transparent',
        // Each resource has a role and a photo in its extendedProps
        resources: people,
        // Each event has a type, a room and the attendees in its extendedProps
        events: meetings
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
            eventContent={({event, timeText}) => (
                <div className={'card ' + event.extendedProps.type}>
                    <div className="card-title">{event.title}</div>
                    <div className="card-meta">{timeText} · {event.extendedProps.room}</div>
                    <div className="card-footer">
                        {event.extendedProps.attendees.map(person => (
                            <img key={person.name} src={person.photo} alt={person.name} title={person.name}/>
                        ))}
                        <span className="card-tag">{event.extendedProps.type}</span>
                    </div>
                </div>
            )}
            resourceLabelContent={({resource}) => (
                <div className="member">
                    <img src={resource.extendedProps.photo} alt=""/>
                    <b>{resource.title}</b>
                    <small>{resource.extendedProps.role}</small>
                </div>
            )}
        />
    );
}

import {EventCalendar, ResourceTimeGrid} from '@event-calendar/vue';
import '@event-calendar/vue/index.css';

const today = new Date();
const people = [/* your resources */];
const meetings = [/* your events */];

    view: 'resourceTimeGridDay',
    height: 'auto',
    slotMinTime: '08:00',
    slotMaxTime: '16:00',
    slotHeight: 40,
    allDaySlot: false,

    events: meetings
};
</script>

<template>
    <EventCalendar :plugins="plugins" :options="options">

                <div class="card-meta">{{ timeText }} · {{ event.extendedProps.room }}</div>

            </div>

                <b>{{ resource.title }}</b>
                <small>{{ resource.extendedProps.role }}</small>
            </div>
        </template>
    </EventCalendar>
</template>

import {Component} from '@angular/core';
import {EventCalendar, EcContent, ResourceTimeGrid, type Calendar} from '@event-calendar/angular';

const today = new Date();
const people: Calendar.ResourceInput[] = [/* your resources */];
const meetings: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `
        <event-calendar [plugins]="plugins" [options]="options">
            <ng-template ecContent="eventContent" let-event="event" let-timeText="timeText">
                <div class="card" [class]="event.extendedProps.type">
                    <div class="card-title">{{ event.title }}</div>
                    <div class="card-meta">{{ timeText }} · {{ event.extendedProps.room }}</div>
                    <div class="card-footer">
                        @for (person of event.extendedProps.attendees; track person.name) {
                            <img [src]="person.photo" [alt]="person.name" [title]="person.name">
                        }
                        <span class="card-tag">{{ event.extendedProps.type }}</span>
                    </div>
                </div>
            </ng-template>
            <ng-template ecContent="resourceLabelContent" let-resource="resource">
                <div class="member">
                    <img [src]="resource.extendedProps.photo" alt="">
                    <b>{{ resource.title }}</b>
                    <small>{{ resource.extendedProps.role }}</small>
                </div>
            </ng-template>
        </event-calendar>`
})
export class App {
    plugins = [ResourceTimeGrid];
    options: Calendar.Options = {
        view: 'resourceTimeGridDay',
        height: 'auto',
        slotMinTime: '08:00',
        slotMaxTime: '16:00',
        slotHeight: 40,
        allDaySlot: false,
        nowIndicator: true,
        buttonText: text => ({...text, today: 'Today'}),
        // The cards have their own colors
        eventBackgroundColor: 'transparent',
        // Each resource has a role and a photo in its extendedProps
        resources: people,
        // Each event has a type, a room and the attendees in its extendedProps
        events: meetings
    };
}
