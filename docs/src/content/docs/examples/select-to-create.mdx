## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, TimeGrid, Interaction} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [TimeGrid, Interaction], {
    view: 'timeGridWeek',
    selectable: true,
    scrollTime: '08:00',
    events,
    select: ({start, end, allDay}) => {
        ec.addEvent({start, end, allDay, title: 'New event', color: '#14b8a6'});
        ec.unselect();
    }
});
```

<TabItem label="React">

```jsx
import {useRef} from 'react';
import {EventCalendar, TimeGrid, Interaction} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [TimeGrid, Interaction];
const events = [/* your events */];

export default function App() {
    const calendar = useRef(null);
    const options = {
        view: 'timeGridWeek',
        selectable: true,
        scrollTime: '08:00',
        events,
        select: ({start, end, allDay}) => {
            calendar.current.addEvent({start, end, allDay, title: 'New event', color: '#14b8a6'});
            calendar.current.unselect();
        }
    };

    return (
        <EventCalendar
            ref={calendar}
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component, viewChild} from '@angular/core';
import {EventCalendar, EcContent, TimeGrid, Interaction, type Calendar} from '@event-calendar/angular';

const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    calendar = viewChild.required(EventCalendar);
    plugins = [TimeGrid, Interaction];
    options: Calendar.Options = {
        view: 'timeGridWeek',
        selectable: true,
        scrollTime: '08:00',
        events,
        select: ({start, end, allDay}) => {
            this.calendar().addEvent({start, end, allDay, title: 'New event', color: '#14b8a6'});
            this.calendar().unselect();
        }
    };
}
```
