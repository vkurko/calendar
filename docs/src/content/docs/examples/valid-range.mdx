The calendar cannot be navigated outside of `validRange`, and the days out of it are disabled.

<Example client:only="svelte" slug="valid-range" />

## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, DayGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const today = new Date();
const addDays = (date, days) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days);
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [DayGrid], {
    view: 'dayGridMonth',
    validRange: {start: addDays(today, -7), end: addDays(today, 21)},
    events
});
```

<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, DayGrid, type Calendar} from '@event-calendar/angular';

const today = new Date();
const addDays = (date, days) => new Date(date.getFullYear(), date.getMonth(), date.getDate() + days);
const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [DayGrid];
    options: Calendar.Options = {
        view: 'dayGridMonth',
        validRange: {start: addDays(today, -7), end: addDays(today, 21)},
        events
    };
}
```
