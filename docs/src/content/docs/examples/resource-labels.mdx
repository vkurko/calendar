## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, ResourceTimeline} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const people = [/* your resources */];
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [ResourceTimeline], {
    view: 'resourceTimelineWeek',
    slotMinTime: '08:00',
    slotMaxTime: '20:00',
    slotDuration: '01:00',
    slotWidth: 72,
    // Each resource has a role and a photo in its extendedProps
    resources: people,
    events,
    resourceLabelContent: ({resource}) => ({
        html: `
            <span class="person">
                <img src="${resource.extendedProps.photo}" alt="">
                <span>
                    <b>${resource.title}</b>
                    <small>${resource.extendedProps.role}</small>
                </span>
            </span>
        `
    })
});
```

import {Component} from '@angular/core';
import {EventCalendar, EcContent, ResourceTimeline, type Calendar} from '@event-calendar/angular';

const people: Calendar.ResourceInput[] = [/* your resources */];
const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `
        <event-calendar [plugins]="plugins" [options]="options">
            <ng-template ecContent="resourceLabelContent" let-resource="resource">
                <span class="person">
                    <img [src]="resource.extendedProps.photo" alt="">
                    <span>
                        <b>{{ resource.title }}</b>
                        <small>{{ resource.extendedProps.role }}</small>
                    </span>
                </span>
            </ng-template>
        </event-calendar>`
})
export class App {
    plugins = [ResourceTimeline];
    options: Calendar.Options = {
        view: 'resourceTimelineWeek',
        slotMinTime: '08:00',
        slotMaxTime: '20:00',
        slotDuration: '01:00',
        slotWidth: 72,
        // Each resource has a role and a photo in its extendedProps
        resources: people,
        events
    };
}
