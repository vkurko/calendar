## Code

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
import {createCalendar, ResourceTimeGrid} from '@event-calendar/core';
import '@event-calendar/core/index.css';

const resources = [/* your resources */];
const events = [/* your events */];

const ec = createCalendar(document.getElementById('ec'), [ResourceTimeGrid], {
    view: 'resourceTimeGridDay',
    scrollTime: '08:00',
    resources,
    events
});
```

</TabItem>
<TabItem label="Svelte">

```svelte
<script>
    import {Calendar, ResourceTimeGrid} from '@event-calendar/core';

    let resources = [/* your resources */];
    let events = [/* your events */];
    let options = $state({
        view: 'resourceTimeGridDay',
        scrollTime: '08:00',
        resources,
        events
    });
</script>

<Calendar plugins={[ResourceTimeGrid]} {options} />
```

</TabItem>
<TabItem label="React">

```jsx
import {EventCalendar, ResourceTimeGrid} from '@event-calendar/react';
import '@event-calendar/react/index.css';

const plugins = [ResourceTimeGrid];
const resources = [/* your resources */];
const events = [/* your events */];

export default function App() {
    const options = {
        view: 'resourceTimeGridDay',
        scrollTime: '08:00',
        resources,
        events
    };

    return (
        <EventCalendar
            plugins={plugins}
            {...options}
        />
    );
}
```

<TabItem label="Angular">

```ts
import {Component} from '@angular/core';
import {EventCalendar, EcContent, ResourceTimeGrid, type Calendar} from '@event-calendar/angular';

const resources: Calendar.ResourceInput[] = [/* your resources */];
const events: Calendar.EventInput[] = [/* your events */];

@Component({
    selector: 'app-root',
    imports: [EventCalendar, EcContent],
    template: `<event-calendar [plugins]="plugins" [options]="options" />`
})
export class App {
    plugins = [ResourceTimeGrid];
    options: Calendar.Options = {
        view: 'resourceTimeGridDay',
        scrollTime: '08:00',
        resources,
        events
    };
}
```
