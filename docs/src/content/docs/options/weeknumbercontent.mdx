## Example

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
let options = {
    weekNumberContent: ({week}) => ({html: `Week ${week}`})
};
```

</TabItem>
<TabItem label="Svelte">

```svelte
<Calendar {plugins} {options}>
    {#snippet weekNumberContent({week})}
        Week {week}
    {/snippet}
</Calendar>
```

</TabItem>
<TabItem label="React">

```jsx
<EventCalendar
    plugins={plugins}
    weekNumberContent={({week}) => <>Week {week}</>}
/>
```

</TabItem>
<TabItem label="Vue">

```vue
<EventCalendar :plugins="plugins">
    <template #weekNumberContent="{week}">
        Week {{ week }}
    </template>
</EventCalendar>
```

</TabItem>
<TabItem label="Angular">

```html
<event-calendar [plugins]="plugins" [options]="options">
    <ng-template ecContent="weekNumberContent" let-week="week">
        Week {{ week }}
    </ng-template>
</event-calendar>
```

</TabItem>
<TabItem label="CDN">

    weekNumberContent: ({week}) => ({html: `Week ${week}`})
