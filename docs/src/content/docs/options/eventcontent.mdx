## Example

<Tabs syncKey="framework">
<TabItem label="Vanilla JS">

```js
let options = {
    eventContent: ({event, timeText}) => ({
        html: `<div class="ec-event-time">${timeText}</div><div class="ec-event-title">★ ${event.title}</div>`
    })
};
```

</TabItem>
<TabItem label="Svelte">

```svelte
<Calendar {plugins} {options}>
    {#snippet eventContent({event, timeText})}
        <div class="ec-event-time">{timeText}</div>
        <div class="ec-event-title">★ {event.title}</div>
    {/snippet}
</Calendar>
```

</TabItem>
<TabItem label="React">

```jsx
<EventCalendar
    plugins={plugins}
    eventContent={({event, timeText}) => <>
        <div className="ec-event-time">{timeText}</div>
        <div className="ec-event-title">★ {event.title}</div>
    </>}
/>
```

</TabItem>
<TabItem label="Vue">

```vue
<EventCalendar :plugins="plugins">

        <div class="ec-event-time">{{ timeText }}</div>

    </template>
</EventCalendar>
```

</TabItem>
<TabItem label="Angular">

```html
<event-calendar [plugins]="plugins" [options]="options">

        <div class="ec-event-title">★ {{ event.title }}</div>
    </ng-template>
</event-calendar>
```
